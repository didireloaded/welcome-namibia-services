🏗️ The Senior-Level Tech Stack
Framework: Next.js 14+ (App Router) — SSR for SEO, Server Actions for secure mutations.

Language: TypeScript — Strict mode.

Styling: Tailwind CSS + shadcn/ui (Radix UI primitives) — Accessible, high-performance, and perfectly suited for the complex layout in your image.

Database: PostgreSQL + Prisma ORM — Relational integrity for complex visa/booking workflows.

Auth: NextAuth.js (Auth.js) v5 — JWT strategy, secure session management.

File Uploads: UploadThing or AWS S3 SDK — For secure document handling (passports, medical records).

Form Validation: React Hook Form + Zod — Client and server-side schema validation.

Animation: Framer Motion — For the smooth carousel and tab transitions in your image.

🗄️ 2. Database Schema (Prisma)
This schema handles the full lifecycle of a client's journey: visa application, airport transfer, medical assistance, and hotel booking.

prisma
// prisma/schema.prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model User {
  id            String    @id @default(cuid())
  email         String    @unique
  password      String
  firstName     String
  lastName      String
  phone         String?
  nationality   String
  role          Role      @default(CLIENT)
  applications  VisaApplication[]
  transfers     TransferBooking[]
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
}

enum Role {
  CLIENT
  ADMIN
  CONSULTANT
}

model VisaApplication {
  id            String    @id @default(cuid())
  userId        String
  user          User      @relation(fields: [userId], references: [id])
  visaType      VisaType
  status        AppStatus @default(DRAFT)
  documents     Document[]
  notes         String?
  submittedAt   DateTime?
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
}

enum VisaType {
  WORK_PERMIT
  STUDY_PERMIT
  TOURIST_VISA
  MEDICAL_VISA
}

enum AppStatus {
  DRAFT
  PENDING_REVIEW
  SUBMITTED_TO_EMBASSY
  APPROVED
  REJECTED
}

model TransferBooking {
  id            String    @id @default(cuid())
  userId        String
  user          User      @relation(fields: [userId], references: [id])
  pickupDate    DateTime
  flightNumber  String
  pickupLocation String   // Airport
  dropoffLocation String  // Hotel/Hospital
  passengers    Int
  vehicleType   String    // Economy, Business, VIP
  status        TransferStatus @default(PENDING)
  createdAt     DateTime  @default(now())
}

enum TransferStatus {
  PENDING
  CONFIRMED
  DRIVER_ASSIGNED
  COMPLETED
  CANCELLED
}

model Document {
  id            String    @id @default(cuid())
  applicationId String
  application   VisaApplication @relation(fields: [applicationId], references: [id])
  fileName      String
  fileUrl       String
  fileType      String
  uploadedAt    DateTime  @default(now())
}

⚙️ 4. Backend Logic (Senior-Level Server Actions)
Instead of messy API routes, we use Next.js Server Actions with Zod validation.

typescript
// app/actions/visa.ts
"use server";

import { z } from "zod";
import { db } from "@/lib/db";
import { auth } from "@/lib/auth";
import { revalidatePath } from "next/cache";

const visaApplicationSchema = z.object({
  visaType: z.enum(["WORK_PERMIT", "STUDY_PERMIT", "TOURIST_VISA", "MEDICAL_VISA"]),
  nationality: z.string().min(2),
  notes: z.string().optional(),
});

export async function submitVisaApplication(formData: FormData) {
  const session = await auth();
  if (!session?.user?.id) {
    return { error: "Unauthorized" };
  }

  const validatedFields = visaApplicationSchema.safeParse({
    visaType: formData.get("visaType"),
    nationality: formData.get("nationality"),
    notes: formData.get("notes"),
  });

  if (!validatedFields.success) {
    return { error: "Invalid fields" };
  }

  try {
    const application = await db.visaApplication.create({
      data: {
        userId: session.user.id,
        visaType: validatedFields.data.visaType,
        status: "PENDING_REVIEW",
        notes: validatedFields.data.notes,
      },
    });

    revalidatePath("/dashboard/client");
    return { success: true, applicationId: application.id };
  } catch (error) {
    console.error("Visa application error:", error);
    return { error: "Failed to submit application. Please try again." };
  }
}
